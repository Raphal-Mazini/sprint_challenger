let
    Nomes = List.Transform(
        List.Select(Record.FieldNames(#shared), each Text.StartsWith(_, "Erros em ")),
        each Text.AfterDelimiter(_, "Erros em ")
    ),
    Analisar = (nome as text) =>
        let
            t = try Record.Field(#shared, nome),
            r =
                if t[HasError] then
                    #table({"Coluna", "Mensagem", "Qtd"}, {{"(consulta inteira)", t[Error][Message], null}})
                else
                    let
                        erros = Table.SelectRowsWithErrors(t[Value]),
                        cols = Table.ColumnNames(erros),
                        marcado = Table.TransformColumns(erros, List.Transform(cols, (c) =>
                            {c, (v) =>
                                let x = try v
                                in if x[HasError]
                                   then x[Error][Message] & " | valor: " &
                                        (try Text.From(x[Error][Detail][Value])
                                         otherwise (try Text.From(x[Error][Detail]) otherwise ""))
                                   else null})),
                        comIndice = Table.AddIndexColumn(marcado, "__i"),
                        despiv = Table.UnpivotOtherColumns(comIndice, {"__i"}, "Coluna", "Mensagem"),
                        agrup = Table.Group(despiv, {"Coluna", "Mensagem"}, {{"Qtd", Table.RowCount, Int64.Type}})
                    in
                        agrup
        in
            Table.AddColumn(r, "Consulta", each nome),
    Resultado = Table.Combine(List.Transform(Nomes, Analisar))
in
    Resultado

DEFINE
    MEASURE 'Gestão de Insumos (Main)'[Finalizados NOVO] =
        CALCULATE(
            COUNTROWS('Gestão de Insumos (Main)'),
            REMOVEFILTERS('Calendar'),
            TREATAS(VALUES('Calendar'[Date]), 'Gestão de Insumos (Main)'[Data_Conclusao_Item])
        )
    MEASURE 'Gestão de Insumos (Main)'[% no Prazo NOVO] =
        DIVIDE(
            CALCULATE([Finalizados NOVO], 'Gestão de Insumos (Main)'[Status Prazo Orizon] = "No Prazo"),
            [Finalizados NOVO]
        )

EVALUATE
VAR meses =
    FILTER(
        DISTINCT(SELECTCOLUMNS(ALL('Calendar'[Date]), "Mes", EOMONTH('Calendar'[Date], 0))),
        [Mes] >= DATE(2026, 1, 31) && [Mes] <= EOMONTH(TODAY(), 0)
    )
RETURN
ADDCOLUMNS(
    meses,
    "% Atual", CALCULATE([% no Prazo Orizon], FILTER(ALL('Calendar'[Date]), EOMONTH('Calendar'[Date], 0) = [Mes])),
    "% Novo", CALCULATE([% no Prazo NOVO], FILTER(ALL('Calendar'[Date]), EOMONTH('Calendar'[Date], 0) = [Mes])),
    "Finalizados", CALCULATE([Finalizados NOVO], FILTER(ALL('Calendar'[Date]), EOMONTH('Calendar'[Date], 0) = [Mes]))
)
ORDER BY [Mes]

% no Prazo Orizon = 
var noprazo = CALCULATE(COUNT('Gestão de Insumos (Main)'[Status Prazo Orizon]),'Gestão de Insumos (Main)'[Status Prazo Orizon] = "No Prazo")
var foradoprazo = CALCULATE(COUNT('Gestão de Insumos (Main)'[Status Prazo Orizon]),'Gestão de Insumos (Main)'[Status Prazo Orizon] = "Fora do Prazo")
var total = noprazo + foradoprazo
RETURN noprazo / total

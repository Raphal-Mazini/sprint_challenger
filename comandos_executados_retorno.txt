EVALUATE
SELECTCOLUMNS(
    FILTER(
        INFO.VIEW.MEASURES(),
        CONTAINSSTRING([Name], "Count") || CONTAINSSTRING([Name], "Protocolo") || CONTAINSSTRING([Name], "Entr")
    ),
    "Tabela", [Table],
    "Medida", [Name],
    "Formula", [Expression]
)
ORDER BY [Tabela], [Medida]

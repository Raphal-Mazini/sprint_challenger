 no Prazo Orizon =
VAR total =
    CALCULATE(
        COUNTROWS('Gestão de Insumos (Main)'),
        REMOVEFILTERS('Calendar'),
        TREATAS(VALUES('Calendar'[Date]), 'Gestão de Insumos (Main)'[Data_Conclusao_Item])
    )
VAR noprazo =
    CALCULATE(
        COUNTROWS('Gestão de Insumos (Main)'),
        REMOVEFILTERS('Calendar'),
        TREATAS(VALUES('Calendar'[Date]), 'Gestão de Insumos (Main)'[Data_Conclusao_Item]),
        'Gestão de Insumos (Main)'[Status Prazo Orizon] = "No Prazo"
    )
RETURN DIVIDE(noprazo, total)
